cmake_minimum_required(VERSION 3.23)
project(opennr LANGUAGES C CXX)
add_executable(opennr src/main.cpp)
target_compile_features(opennr PRIVATE cxx_std_20)
target_compile_definitions(opennr PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(opennr PRIVATE external/json)
target_link_libraries(opennr PRIVATE bcrypt)
if(MSVC)
  target_compile_options(opennr PRIVATE /utf-8 /W4)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(opennr_text_tests tests/native/text_tests.cpp)
  target_compile_features(opennr_text_tests PRIVATE cxx_std_20)
  target_compile_definitions(opennr_text_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_include_directories(opennr_text_tests PRIVATE src)
  target_link_libraries(opennr_text_tests PRIVATE bcrypt)
  if(MSVC)
    target_compile_options(opennr_text_tests PRIVATE /utf-8 /W4)
  endif()
  add_test(NAME opennr_text_views COMMAND opennr_text_tests)
endif()

set(OPENNR_NATIVE_BENCHMARKS_DEFAULT OFF)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/local/native-dependencies/manifest.json")
  set(OPENNR_NATIVE_BENCHMARKS_DEFAULT ON)
endif()
option(OPENNR_BUILD_NATIVE_BENCHMARKS "Build native D3D11 and D3D12 benchmark harnesses" ${OPENNR_NATIVE_BENCHMARKS_DEFAULT})
if(OPENNR_BUILD_NATIVE_BENCHMARKS)
  add_subdirectory(benchmarks/native)
endif()
